-- ToggleBus — bus event client-side, pengganti _G.ToggleBus
-- Pakai:
--   ToggleBus.toggle("Parry", true)
--   ToggleBus.onToggle(function(feature, state) ... end)

local ToggleBus = {}
local signal = Instance.new("BindableEvent")

function ToggleBus.toggle(feature: string, state: boolean)
	signal:Fire(feature, state)
end

function ToggleBus.onToggle(callback: (string, boolean) -> ())
	return signal.Event:Connect(callback)
end

return ToggleBus--!strict
-- ParryAPI — state parry per pemain. HANYA untuk server.
-- Karena ada di ServerStorage, client tidak bisa baca/modifikasi.

type State = { lastParry: number, activeUntil: number }

local ParryAPI = {}
local parryState: {[Player]: State} = {}
local lastRemoteAt: {[Player]: number} = {}

local REMOTE_RATE = 0.1 -- anti-spam remote

local function getState(p: Player): State
	if not parryState[p] then
		parryState[p] = { lastParry = 0, activeUntil = 0 }
	end
	return parryState[p]
end

function ParryAPI.tryParry(player: Player): boolean
	local now = os.clock()

	-- Rate limit
	if now - (lastRemoteAt[player] or 0) < REMOTE_RATE then return false end
	lastRemoteAt[player] = now

	-- Karakter hidup?
	local char = player.Character
	if not char then return false end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return false end

	-- Cooldown
	local state = getState(player)
	if now - state.lastParry < Config.Parry.COOLDOWN then return false end

	-- Terima parry
	state.lastParry = now
	state.activeUntil = now + Config.Parry.WINDOW
	return true
end

function ParryAPI.isParrying(player: Player, now: number?): boolean
	local state = parryState[player]
	if not state then return false end
	return (now or os.clock()) <= state.activeUntil
end

Players.PlayerRemoving:Connect(function(p)
	parryState[p] = nil
	lastRemoteAt[p] = nil
end)

return ParryAPI--!strict

local lastAttackAt: {[Player]: number} = {}

local function getHRP(p: Player): BasePart?
	local c = p.Character
	if not c then return nil end
	return c:FindFirstChild("HumanoidRootPart") :: BasePart?
end

local function tryDamage(attacker: Player, target: Player, damage: number): string
	local targetHRP = getHRP(target)
	local attackerHRP = getHRP(attacker)
	if not targetHRP or not attackerHRP then return "invalid" end

	local dist = (targetHRP.Position - attackerHRP.Position).Magnitude
	if dist > Config.Combat.ATTACK_RANGE then return "invalid" end

	local now = os.clock()

	if ParryAPI.isParrying(target, now) then
		if dist <= Config.Parry.MAX_DISTANCE then
			print(("[Parry] %s parry serangan %s"):format(target.Name, attacker.Name))
			return "parried"
		end
	end

	local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
	if hum then hum:TakeDamage(damage) end
	return "hit"
end

AttackRequest.OnServerEvent:Connect(function(player, target)
	-- Validasi tipe (client bisa kirim apa saja)
	if typeof(target) ~= "Instance" then return end
	if not target:IsA("Model") then return end

	-- Rate limit
	local now = os.clock()
	if now - (lastAttackAt[player] or 0) < Config.Combat.ATTACK_COOLDOWN then return end
	lastAttackAt[player] = now

	-- Harus karakter pemain lain
	local targetPlayer = Players:GetPlayerFromCharacter(target)
	if not targetPlayer or targetPlayer == player then return end

	local hum = target:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return end

	tryDamage(player, targetPlayer, Config.Combat.BASE_DAMAGE)
end)

Players.PlayerRemoving:Connect(function(p)
	lastAttackAt[p] = nil
end)--!strict

local function createBillboard(part: BasePart): BillboardGui
	local bb = Instance.new("BillboardGui")
	bb.Name = "GenProgress"
	bb.Size = UDim2.fromOffset(120, 30)
	bb.StudsOffsetWorldSpace = Vector3.new(0, 3.5, 0)
	bb.AlwaysOnTop = true
	bb.MaxDistance = 120
	bb.Adornee = part
	bb.Parent = part

	local bg = Instance.new("Frame")
	bg.Size = UDim2.fromScale(1, 0.5)
	bg.Position = UDim2.fromScale(0, 0.5)
	bg.AnchorPoint = Vector2.new(0, 0.5)
	bg.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
	bg.BorderSizePixel = 0
	bg.Parent = bb
	Instance.new("UICorner", bg).CornerRadius = UDim.new(1, 0)

	local fill = Instance.new("Frame")
	fill.Name = "Fill"
	fill.Size = UDim2.fromScale(0, 1)
	fill.BackgroundColor3 = Config.GUI.COLOR_ACCENT
	fill.BorderSizePixel = 0
	fill.Parent = bg
	Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

	local label = Instance.new("TextLabel")
	label.Name = "Label"
	label.Size = UDim2.fromScale(1, 0.4)
	label.Position = UDim2.fromScale(0, 0)
	label.AnchorPoint = Vector2.new(0, 1)
	label.BackgroundTransparency = 1
	label.Font = Config.GUI.FONT
	label.TextColor3 = Config.GUI.COLOR_TEXT
	label.TextStrokeTransparency = 0.5
	label.TextSize = 12
	label.Text = "0%"
	label.Parent = bb

	local function update()
		local progress = part:GetAttribute("Progress") or 0
		local active   = part:GetAttribute("Active") or false
		local pct = math.clamp(progress / Config.Generator.FINISH_TIME, 0, 1)

		TweenService:Create(fill, TweenInfo.new(0.2), {
			Size = UDim2.fromScale(pct, 1)
		}):Play()

		label.Text = ("%d%%"):format(math.floor(pct * 100))
		bb.Enabled = active or (pct > 0 and pct < 1)

		if pct >= 1 then
			task.delay(1.5, function()
				if bb.Parent then bb.Enabled = false end
			end)
		end
	end

	part:GetAttributeChangedSignal("Progress"):Connect(update)
	part:GetAttributeChangedSignal("Active"):Connect(update)
	update()

	return bb
end

local function setupPrompt(part: BasePart)
	if part:FindFirstChildOfClass("ProximityPrompt") then return end

	local prompt = Instance.new("ProximityPrompt")
	prompt.ActionText = "Aktifkan"
	prompt.ObjectText = "Generator"
	prompt.HoldDuration = 0
	prompt.MaxActivationDistance = 10
	prompt.RequiresLineOfSight = false
	prompt.Parent = part

	prompt.Triggered:Connect(function()
		interact(part, instantMode and "instant" or "start")
	end)
	prompt.TriggerEnded:Connect(function()
		if not instantMode then interact(part, "stop") end
	end)

	createBillboard(part)
end

local enabled = false
local activeHighlights: {[Instance]: Highlight} = {}

local function addHighlight(instance: Instance, color: Color3)
	if activeHighlights[instance] then return end
	local hl = Instance.new("Highlight")
	hl.Adornee = instance
	hl.FillColor = color
	hl.OutlineColor = Color3.new(1, 1, 1)
	hl.FillTransparency = 0.5
	hl.OutlineTransparency = 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.Parent = instance
	activeHighlights[instance] = hl
end

local function clearAll()
	for inst, hl in pairs(activeHighlights) do
		hl:Destroy()
		activeHighlights[inst] = nil
	end
end

local function scanAndTag()
	for inst, hl in pairs(activeHighlights) do
		if not inst.Parent then
			hl:Destroy()
			activeHighlights[inst] = nil
		end
	end
	for _, obj in ipairs(Workspace:GetDescendants()) do
		local color = Config.DebugHighlight.COLORS[obj.Name]
		if color and (obj:IsA("Model") or obj:IsA("BasePart")) then
			addHighlight(obj, color)
		end
	end
end

ToggleBus.onToggle(function(feature, state)
	if feature == "ESP" then
		enabled = state
		if enabled then scanAndTag() else clearAll() end
	end
end)

task.spawn(function()
	while true do
		task.wait(2)
		if enabled then scanAndTag() end
	end
end)--!strict

type State = { lastParry: number, activeUntil: number }
local parryState: {[Player]: State} = {}
local lastRemoteAt: {[Player]: number} = {}

local REMOTE_RATE = 0.1 -- server tolak jika < 0.1s

local function getState(p: Player): State
	if not parryState[p] then
		parryState[p] = { lastParry = 0, activeUntil = 0 }
	end
	return parryState[p]
end

ParryRequest.OnServerEvent:Connect(function(player)
	-- Rate limit remote
	local now = os.clock()
	if now - (lastRemoteAt[player] or 0) < REMOTE_RATE then return end
	lastRemoteAt[player] = now

	-- Karakter hidup?
	local char = player.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then return end

	-- Cooldown
	local state = getState(player)
	if now - state.lastParry < Config.Parry.COOLDOWN then return end

	state.lastParry = now
	state.activeUntil = now + Config.Parry.WINDOW
end)

end

------------------------------------------------------------
-- API: tryDamage(attacker, target, damage, fromPos)
-- Return: "parried" | "hit" | "invalid"
------------------------------------------------------------
local function tryDamage(attacker: Player, target: Player, damage: number, fromPos: Vector3?)
	local targetHRP = getHRP(target)
	local attackerHRP = getHRP(attacker)
	if not targetHRP or not attackerHRP then return "invalid" end

	local originPos = fromPos or attackerHRP.Position

	-- Range check
	if (targetHRP.Position - originPos).Magnitude > Config.Combat.ATTACK_RANGE then
		return "invalid"
	end

	local now = os.clock()

	-- Parry?
	if ParryAPI.isParrying(target, now) then
		if (targetHRP.Position - originPos).Magnitude <= Config.Parry.MAX_DISTANCE then
			print(("[Parry] %s parry serangan %s"):format(target.Name, attacker.Name))
			return "parried"
		end
	end

	-- Damage
	local hum = target.Character and target.Character:FindFirstChildOfClass("Humanoid")
	if hum then
		hum:TakeDamage(damage)
	end
	return "hit"
end

_G.TryDamage = tryDamage

GeneratorAction.OnServerEvent:Connect(function(player, action, part)
	-- Validasi tipe
	if typeof(action) ~= "string" then return end
	if typeof(part) ~= "Instance" then return end
	if not part:IsA("BasePart") then return end

	local state = generators[part]
	if not state then return end

	-- Rate limit
	local now = os.clock()
	if now - (lastActionAt[player] or 0) < REMOTE_RATE then return end
	lastActionAt[player] = now

	-- Range check
	if not inRange(player, part, Config.Generator.INTERACT_RANGE) then return end

	if action == "start" then
		if state.active then return end
		state.active = true
		state.lastInteractor = player
		part:SetAttribute("Active", true)

	elseif action == "stop" then
		if state.lastInteractor ~= player then return end
		state.active = false
		state.lastInteractor = nil
		part:SetAttribute("Active", false)

	elseif action == "instant" then
		-- ⚡ HANYA di Studio + hanya dev
		if not RunService:IsStudio() then return end
		if not isDev(player) then return end

		state.progress = Config.Generator.FINISH_TIME
		state.active = false
		state.lastInteractor = nil
		part:SetAttribute("Progress", Config.Generator.FINISH_TIME)
		part:SetAttribute("Active", false)
		print(("[Generator] INSTANT oleh %s"):format(player.Name))
	end
end)

task.spawn(function()
	while true do
		task.wait(Config.Generator.TICK_RATE)

		for part, state in pairs(generators) do
			if not part.Parent then
				generators[part] = nil
				continue
			end

			if state.active and state.lastInteractor then
				-- Interaktor masih dekat?
				if not inRange(state.lastInteractor, part, Config.Generator.INTERACT_RANGE * 1.5) then
					state.active = false
					state.lastInteractor = nil
					part:SetAttribute("Active", false)
					continue
				end

				state.progress = math.min(
					state.progress + Config.Generator.PROGRESS_PER_TICK,
					Config.Generator.FINISH_TIME
				)
				part:SetAttribute("Progress", state.progress)

				if state.progress >= Config.Generator.FINISH_TIME then
					state.active = false
					state.lastInteractor = nil
					part:SetAttribute("Active", false)
					print(("[Generator] %s selesai!"):format(part:GetFullName()))
					-- TODO: reward, buka pintu, dsb.
				end
			end
		end
	end
end)--!strict

local enabled = false
local activeHighlights: {[Instance]: Highlight} = {}

local function addHighlight(instance: Instance, color: Color3)
	if activeHighlights[instance] then return end
	local hl = Instance.new("Highlight")
	hl.Adornee = instance
	hl.FillColor = color
	hl.OutlineColor = Color3.new(1, 1, 1)
	hl.FillTransparency = 0.5
	hl.OutlineTransparency = 0
	hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	hl.Parent = instance
	activeHighlights[instance] = hl
end

local function clearAll()
	for inst, hl in pairs(activeHighlights) do
		hl:Destroy()
		activeHighlights[inst] = nil
	end
end

local function scanAndTag()
	-- Bersihkan yang parent sudah hilang
	for inst, hl in pairs(activeHighlights) do
		if not inst.Parent then
			hl:Destroy()
			activeHighlights[inst] = nil
		end
	end

	for _, obj in ipairs(Workspace:GetDescendants()) do
		local color = Config.DebugHighlight.COLORS[obj.Name]
		if color and (obj:IsA("Model") or obj:IsA("BasePart")) then
			addHighlight(obj, color)
		end
	end
end
